const SPREADSHEET_ID = "ここにスプレッドシートIDを省略なしで入れる";
const SHEET_NAME = "Plain text i&u online album管理表";
const ADMIN_KEY = "iu2026";
function doGet(e){const a=e.parameter.action||"album",c=e.parameter.callback;if(a==="add")return addAlbum(e,c);if(a==="list")return listAlbums(e,c);if(a==="update")return updateAlbum(e,c);if(a==="delete")return deleteAlbum(e,c);return getAlbum(e,c)}
function getSheet(){return SpreadsheetApp.openById(SPREADSHEET_ID).getSheetByName(SHEET_NAME)}
function requireAdmin(e,c){if(e.parameter.key!==ADMIN_KEY)return output({ok:false,error:"管理キーが違います"},c);return null}
function ensureHeaders(){const s=getSheet();let h=s.getRange(1,1,1,Math.max(s.getLastColumn(),1)).getValues()[0];["slug","title","deliveryDate","limitDate","folderId","password","customer","type"].forEach(x=>{if(!h.includes(x)){s.getRange(1,s.getLastColumn()+1).setValue(x);h.push(x)}});return h}
function getRows(){const s=getSheet(),h=ensureHeaders(),v=s.getDataRange().getValues(),r=v.slice(1).map((row,i)=>{const o={rowNumber:i+2};h.forEach((x,j)=>o[x]=row[j]);return o});return{sheet:s,headers:h,rows:r}}
function appendByHeaders(d){const s=getSheet(),h=ensureHeaders();s.appendRow(h.map(x=>d[x]||""))}
function writeByHeaders(n,d){const s=getSheet(),h=ensureHeaders();h.forEach((x,i)=>{if(d[x]!==undefined)s.getRange(n,i+1).setValue(d[x])})}
function addAlbum(e,c){const auth=requireAdmin(e,c);if(auth)return auth;const slug=e.parameter.slug;if(!slug)return output({ok:false,error:"slugがありません"},c);const {rows}=getRows();if(rows.some(r=>String(r.slug)===String(slug)))return output({ok:false,error:"同じslugがすでにあります"},c);appendByHeaders({slug,title:e.parameter.title||"",deliveryDate:e.parameter.deliveryDate||"",limitDate:e.parameter.limitDate||"",folderId:e.parameter.folderId||"",password:e.parameter.password||"",customer:e.parameter.customer||"",type:e.parameter.type||""});return output({ok:true,slug},c)}
function listAlbums(e,c){const auth=requireAdmin(e,c);if(auth)return auth;const {rows}=getRows();const albums=rows.filter(r=>r.slug).map(r=>{delete r.rowNumber;return r});return output({ok:true,albums},c)}
function updateAlbum(e,c){const auth=requireAdmin(e,c);if(auth)return auth;const {rows}=getRows(),a=rows.find(r=>String(r.slug)===String(e.parameter.slug));if(!a)return output({ok:false,error:"アルバムが見つかりません"},c);writeByHeaders(a.rowNumber,{title:e.parameter.title||a.title,folderId:e.parameter.folderId||a.folderId,password:e.parameter.password||a.password,limitDate:e.parameter.limitDate||a.limitDate,customer:e.parameter.customer||a.customer,type:e.parameter.type||a.type});return output({ok:true,slug:e.parameter.slug},c)}
function deleteAlbum(e,c){const auth=requireAdmin(e,c);if(auth)return auth;const {sheet,rows}=getRows(),a=rows.find(r=>String(r.slug)===String(e.parameter.slug));if(!a)return output({ok:false,error:"アルバムが見つかりません"},c);sheet.deleteRow(a.rowNumber);return output({ok:true,slug:e.parameter.slug},c)}
function getAlbum(e,c){const slug=e.parameter.slug||"sample",{rows}=getRows(),a=rows.find(x=>String(x.slug)===String(slug));if(!a)return output({error:"album not found"},c);const folder=DriveApp.getFolderById(a.folderId),files=folder.getFiles(),photos=[];while(files.hasNext()){const f=files.next(),id=f.getId();photos.push({name:f.getName(),imageUrl:"https://drive.google.com/thumbnail?id="+id+"&sz=w1000",largeUrl:"https://drive.google.com/thumbnail?id="+id+"&sz=w2000",mobileUrl:"https://drive.google.com/thumbnail?id="+id+"&sz=w1600",downloadUrl:"https://drive.google.com/file/d/"+id+"/view?usp=drivesdk"})}delete a.rowNumber;a.photos=photos;return output(a,c)}
function output(d,c){const j=JSON.stringify(d);if(c)return ContentService.createTextOutput(c+"("+j+");").setMimeType(ContentService.MimeType.JAVASCRIPT);return ContentService.createTextOutput(j).setMimeType(ContentService.MimeType.JSON)}
